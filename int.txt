import React, { useState } from "react";

export default function TimeZone() {
  const [fromZone, setFromZone] = useState("Asia/Kolkata");
  const [toZone, setToZone] = useState("Europe/Berlin");
  const [inputTime, setInputTime] = useState("");
  const [convertedTime, setConvertedTime] = useState("");

  const convertTime = async () => {
    try {
      // Fetch both zones
      const fromRes = await fetch(`https://worldtimeapi.org/api/timezone/${fromZone}`);
      const toRes = await fetch(`https://worldtimeapi.org/api/timezone/${toZone}`);
      const fromData = await fromRes.json();
      const toData = await toRes.json();

      // Extract UTC offsets (e.g. +05:30 -> 5.5 hours)
      const fromOffset = parseFloat(fromData.utc_offset.slice(0, 3)) + 
        (parseInt(fromData.utc_offset.slice(4, 6)) / 60) * (fromData.utc_offset.startsWith("-") ? -1 : 1);
      const toOffset = parseFloat(toData.utc_offset.slice(0, 3)) + 
        (parseInt(toData.utc_offset.slice(4, 6)) / 60) * (toData.utc_offset.startsWith("-") ? -1 : 1);

      // Convert input time to Date in source timezone
      const [hours, minutes] = inputTime.split(":").map(Number);
      const inputDate = new Date();
      inputDate.setHours(hours);
      inputDate.setMinutes(minutes);

      // Convert to UTC
      const utcTime = inputDate.getTime() - fromOffset * 60 * 60 * 1000;

      // Convert to target timezone
      const targetTime = new Date(utcTime + toOffset * 60 * 60 * 1000);

      setConvertedTime(targetTime.toLocaleTimeString());
    } catch (error) {
      console.error("Error:", error);
      setConvertedTime("Error converting time");
    }
  };

  return (
    <div className="flex flex-col items-center p-6 bg-gray-100 min-h-screen">
      <h2 className="text-2xl font-semibold mb-4">🌍 Live Time Zone Converter</h2>

      <div className="flex flex-col gap-3 bg-white shadow p-4 rounded-lg w-80">
        <label>From Timezone:</label>
        <input
          type="text"
          value={fromZone}
          onChange={(e) => setFromZone(e.target.value)}
          className="border p-2 rounded"
          placeholder="Asia/Kolkata"
        />

        <label>To Timezone:</label>
        <input
          type="text"
          value={toZone}
          onChange={(e) => setToZone(e.target.value)}
          className="border p-2 rounded"
          placeholder="Europe/Berlin"
        />

        <label>Time in Source:</label>
        <input
          type="time"
          value={inputTime}
          onChange={(e) => setInputTime(e.target.value)}
          className="border p-2 rounded"
        />

        <button
          onClick={convertTime}
          className="bg-blue-500 text-white p-2 rounded hover:bg-blue-600 mt-2"
        >
          Convert
        </button>
      </div>

      {convertedTime && (
        <div className="mt-4 text-lg font-medium">
          Converted Time in {toZone}: <span className="font-bold">{convertedTime}</span>
        </div>
      )}
    </div>
  );
}
